    /* ArrayList grows automatically, so this stack has no fixed event limit. */
    private static class DynamicMovementStack {
        private final ArrayList<Movement> items = new ArrayList<>();

        public synchronized void push(Movement movement) {
            items.add(movement);
        }

        public synchronized Movement pop() {
            if (items.isEmpty()) return null;
            return items.remove(items.size() - 1);
        }

        public synchronized List<Movement> snapshot() {
            return new ArrayList<>(items);
        }

        public synchronized void replaceAll(List<Movement> movements) {
            items.clear();
            items.addAll(movements);
        }

        public synchronized void clear() {
            items.clear();
        }

        public synchronized int size() {
            return items.size();
        }

        public synchronized boolean isEmpty() {
            return items.isEmpty();
        }
    }

    /* Each event represents a key going down OR coming back up. */

        private final int keyCode;
        private final String key;
        private final boolean pressed;

            this.keyCode = keyCode;
            this.key = key;
            this.pressed = pressed;
            this.elapsedNanos = elapsedNanos;
        }
    }

    private void installKeyboardHook() {
        Logger.getLogger(GlobalScreen.class.getPackage().getName()).setLevel(Level.OFF);
        try {
            if (!GlobalScreen.isNativeHookRegistered()) GlobalScreen.registerNativeHook();
            GlobalScreen.addNativeKeyListener(this);
            hookRegistered = true;
            hookStatus.setText("Keyboard hook: registered");
        } catch (NativeHookException | RuntimeException ex) {
            hookRegistered = false;
            hookStatus.setText("Keyboard hook: ERROR");
            JOptionPane.showMessageDialog(this, "Keyboard capture could not start:\n" + ex.getMessage(),
                    "Keyboard Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private String getKey(int code) {
        switch (code) {
            case NativeKeyEvent.VC_W: return "W";
            case NativeKeyEvent.VC_A: return "A";
            case NativeKeyEvent.VC_S: return "S";
            case NativeKeyEvent.VC_D: return "D";
            case NativeKeyEvent.VC_J: return "J";
            case NativeKeyEvent.VC_R: return "R";
            case NativeKeyEvent.VC_Q: return "Q";
            case NativeKeyEvent.VC_F: return "F";
            default: return "";
        }
    }

    private int awtKeyCode(String key) {
        switch (key) {
            case "W": return KeyEvent.VK_W;
            case "A": return KeyEvent.VK_A;
            case "S": return KeyEvent.VK_S;
            case "D": return KeyEvent.VK_D;
            case "J": return KeyEvent.VK_J;
            case "R": return KeyEvent.VK_R;
            case "Q": return KeyEvent.VK_Q;
            case "F": return KeyEvent.VK_F;
            default: return KeyEvent.VK_UNDEFINED;
        }
    }

    @Override
    public void nativeKeyPressed(NativeKeyEvent event) {
        int code = event.getKeyCode();
        if (code == NativeKeyEvent.VC_F9) {
            SwingUtilities.invokeLater(this::toggleRecording);
            return;
        }
        if (code == NativeKeyEvent.VC_F10) {
            SwingUtilities.invokeLater(() -> playMacro(false));
            return;
        }
        if (code == NativeKeyEvent.VC_F11) {
            SwingUtilities.invokeLater(this::stopMacro);
            return;
        }
        String key = getKey(code);
        if (key.isEmpty()) return;
        boolean firstPress;
        synchronized (inspectorHeldKeys) {
            firstPress = inspectorHeldKeys.add(code);
        }
        if (firstPress) updateKeyboardInspector(key, true);
        if (!recordingActive.get()) return;
        synchronized (heldNativeKeys) {
            if (!heldNativeKeys.add(code)) return;
        }
        addMovement(new Movement(awtKeyCode(key), key, true, System.nanoTime() - recordingStart));
    }

    @Override
    public void nativeKeyReleased(NativeKeyEvent event) {
        int code = event.getKeyCode();
        String key = getKey(code);
        if (key.isEmpty()) return;
        boolean wasInspectorHeld;
        synchronized (inspectorHeldKeys) {
            wasInspectorHeld = inspectorHeldKeys.remove(code);
        }
        if (wasInspectorHeld) updateKeyboardInspector(key, false);
        boolean wasHeld;
        synchronized (heldNativeKeys) {
            wasHeld = heldNativeKeys.remove(code);
        }
        if (wasHeld && recordingActive.get()) {
            addMovement(new Movement(awtKeyCode(key), key, false, System.nanoTime() - recordingStart));
        }
    }

    @Override
    public void nativeKeyTyped(NativeKeyEvent event) { }

    private void updateKeyboardInspector(String key, boolean pressed) {
        SwingUtilities.invokeLater(() -> {
            List<String> held = new ArrayList<>();
            synchronized (inspectorHeldKeys) {
                for (int code : inspectorHeldKeys) held.add(getKey(code));
            }
            java.util.Collections.sort(held);
            heldKeysLabel.setText(held.isEmpty() ? "NONE" : String.join(" + ", held));

            String event = String.format("%-2s  %s  %s", key,
                    pressed ? "PRESSED " : "RELEASED", recordingActive.get() ? "RECORDING" : "MONITORING");
            inspectorModel.addElement(event);
            while (inspectorModel.size() > 100) inspectorModel.remove(0);
            inspectorList.ensureIndexIsVisible(inspectorModel.size() - 1);
        });
    }

    private void addMovement(Movement movement) {
        movements.push(movement);
        SwingUtilities.invokeLater(() -> {
            addLogLine(movement);
            count.setText(movements.size() + " events");
            lastAction.setText(movement.key + (movement.pressed ? " DOWN" : " UP"));
            refreshButtons();
        });
    }

    private void addLogLine(Movement movement) {
        long millis = movement.elapsedNanos / 1_000_000;
        String event = movement.pressed ? "KEY DOWN" : "KEY UP";
        logModel.addElement(String.format("%-2s  %-8s  %02d:%02d.%03d", movement.key, event,
                millis / 60000, (millis / 1000) % 60, millis % 1000));
        activityList.ensureIndexIsVisible(logModel.size() - 1);
        updateTimer();
    }

    private void toggleRecording() {
        if (playing.get()) return;
        if (recordingActive.get()) {
            stopRecording();
        } else {
            synchronized (heldNativeKeys) { heldNativeKeys.clear(); }
            recordingStart = System.nanoTime();
            recordingActive.set(true);
            status.setText("RECORDING");
            recordButton.setText("■ STOP RECORDING");
            recordButton.setBackground(new Color(130, 20, 30));
        }
        refreshButtons();
    }

    private void stopRecording() {
        recordingActive.set(false);
        long elapsed = System.nanoTime() - recordingStart;
        List<Integer> keysToRelease = new ArrayList<>();
        synchronized (heldNativeKeys) {
            keysToRelease.addAll(heldNativeKeys);
            heldNativeKeys.clear();
        }
        for (int nativeCode : keysToRelease) {
            String key = getKey(nativeCode);
            if (!key.isEmpty()) addMovement(new Movement(awtKeyCode(key), key, false, elapsed));
        }
        status.setText("STANDBY");
        recordButton.setText("● START RECORDING");
        recordButton.setBackground(RED);
        refreshButtons();
    }

    private void updateTimer() {
        if (!recordingActive.get()) return;
        long elapsed = (System.nanoTime() - recordingStart) / 1_000_000;
        timerLabel.setText(String.format("%02d:%02d.%03d", elapsed / 60000,
                (elapsed / 1000) % 60, elapsed % 1000));
    }

    private void playMacro(boolean countdown) {
        if (recordingActive.get()) stopRecording();
        if (!playing.compareAndSet(false, true)) return;
        List<Movement> copy = movements.snapshot();
        if (copy.isEmpty()) {
            playing.set(false);
            JOptionPane.showMessageDialog(this, "Record or load movements first.");
            refreshButtons();
            return;
        }
        stopPlayback.set(false);
        status.setText("PLAYING");
        refreshButtons();
        final int repeats = (Integer) repeatSpinner.getValue();
        final double speed = speedSlider.getValue() / 100.0;
        final boolean startMusic = musicWithMacro.isSelected();
        playbackThread = new Thread(() -> runPlayback(copy, countdown, repeats, speed, startMusic), "HeroMacroPlayback");
        playbackThread.setDaemon(true);
        playbackThread.start();
    }

    private void runPlayback(List<Movement> copy, boolean countdown, int repeats, double speed, boolean startMusic) {
        Robot robot = null;
        try {
            if (countdown) {
                for (int i = 3; i > 0 && !closing && !stopPlayback.get(); i--) {
                    final int seconds = i;
                    SwingUtilities.invokeLater(() -> status.setText("SWITCH IN " + seconds));
                    Thread.sleep(1000);
                }
            }
            robot = new Robot();
            if (startMusic) SwingUtilities.invokeLater(music::playOrResume);
            for (int pass = 0; pass < repeats && !closing && !stopPlayback.get(); pass++) {
                long previous = 0;
                for (Movement movement : copy) {
                    if (closing || stopPlayback.get()) break;
                    long delay = movement.elapsedNanos - previous;
                    if (delay > 0) sleepNanos((long) (delay / speed));
                    if (closing || stopPlayback.get()) break;
                    if (movement.pressed) {
                        robot.keyPress(movement.keyCode);
                        synchronized (playbackHeldKeys) { playbackHeldKeys.add(movement.keyCode); }
                    } else {
                        robot.keyRelease(movement.keyCode);
                        synchronized (playbackHeldKeys) { playbackHeldKeys.remove(movement.keyCode); }
                    }
                    previous = movement.elapsedNanos;
                }
                releasePlaybackKeys(robot);
            }
        } catch (Exception ex) {
            SwingUtilities.invokeLater(() -> JOptionPane.showMessageDialog(this,
                    "Playback error: " + ex.getMessage(), "Playback Error", JOptionPane.ERROR_MESSAGE));
        } finally {
            if (robot != null) releasePlaybackKeys(robot);
            if (startMusic) SwingUtilities.invokeLater(music::stop);
            playing.set(false);
            playbackThread = null;
            SwingUtilities.invokeLater(() -> {
                status.setText("STANDBY");
                refreshButtons();
            });
        }
    }

    private void sleepNanos(long nanos) throws InterruptedException {
        long end = System.nanoTime() + nanos;
        while (!closing && !stopPlayback.get()) {
            long remaining = end - System.nanoTime();
            if (remaining <= 0) return;
            long chunk = Math.min(remaining, 25_000_000L);
            long millis = chunk / 1_000_000;
            int extraNanos = (int) (chunk % 1_000_000);
            Thread.sleep(millis, extraNanos);
        }
    }

    private void releasePlaybackKeys(Robot robot) {
        synchronized (playbackHeldKeys) {
            for (int keyCode : playbackHeldKeys) robot.keyRelease(keyCode);
            playbackHeldKeys.clear();
        }
    }

    private void stopMacro() {
        boolean wasPlaying = playing.get();
        boolean wasRecording = recordingActive.get();
        if (!wasPlaying && !wasRecording) return;
        stopPlayback.set(true);
        if (wasRecording) stopRecording();
        if (wasPlaying) status.setText("STOPPING");
        refreshButtons();
    }

    private void saveMacro() {
        if (recordingActive.get() || playing.get()) {
            JOptionPane.showMessageDialog(this, "Stop recording and playback before saving.");
            return;
        }
        JFileChooser chooser = new JFileChooser();
        chooser.setDialogTitle("Save Hero Macro");
        chooser.setSelectedFile(new File("hero-macro.ham"));
        if (chooser.showSaveDialog(this) != JFileChooser.APPROVE_OPTION) return;
        File file = chooser.getSelectedFile();
        if (!file.getName().toLowerCase().endsWith(".ham")) file = new File(file.getParentFile(), file.getName() + ".ham");
        List<Movement> copy = movements.snapshot();
        try (DataOutputStream out = new DataOutputStream(new BufferedOutputStream(new FileOutputStream(file)))) {
            out.writeUTF(FILE_MAGIC);
            out.writeInt(FILE_VERSION);
            out.writeInt(copy.size());
            for (Movement movement : copy) {
                out.writeInt(movement.keyCode);
                out.writeUTF(movement.key);
                out.writeBoolean(movement.pressed);
                out.writeLong(movement.elapsedNanos);
            }
            JOptionPane.showMessageDialog(this, "Saved " + copy.size() + " events to:\n" + file.getAbsolutePath());
        } catch (IOException ex) {
            JOptionPane.showMessageDialog(this, "Could not save macro: " + ex.getMessage(), "Save Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void loadMacro() {
        if (recordingActive.get() || playing.get()) {
            JOptionPane.showMessageDialog(this, "Stop recording and playback before loading.");
            return;
        }
        JFileChooser chooser = new JFileChooser();
        chooser.setDialogTitle("Load Hero Macro");
        if (chooser.showOpenDialog(this) != JFileChooser.APPROVE_OPTION) return;
        File file = chooser.getSelectedFile();
        List<Movement> loaded = new ArrayList<>();
        try (DataInputStream in = new DataInputStream(new BufferedInputStream(new FileInputStream(file)))) {
            String magic = in.readUTF();
            int version = in.readInt();
            int amount = in.readInt();
            if (!FILE_MAGIC.equals(magic) || version != FILE_VERSION || amount < 0 || amount > 2_000_000) {
                throw new IOException("This is not a supported Hero Macro file.");
            }
            long previousTime = -1;
            for (int i = 0; i < amount; i++) {
                int keyCode = in.readInt();
                String key = in.readUTF();
                boolean pressed = in.readBoolean();
                long elapsed = in.readLong();
                if (awtKeyCode(key) == KeyEvent.VK_UNDEFINED || awtKeyCode(key) != keyCode
                        || elapsed < previousTime || elapsed < 0) {
                    throw new IOException("The macro contains invalid movement data.");
                }
                loaded.add(new Movement(keyCode, key, pressed, elapsed));
                previousTime = elapsed;
            }
            movements.replaceAll(loaded);
            rebuildLog();
            JOptionPane.showMessageDialog(this, "Loaded " + loaded.size() + " events from:\n" + file.getAbsolutePath());
        } catch (IOException ex) {
            JOptionPane.showMessageDialog(this, "Could not load macro: " + ex.getMessage(), "Load Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void chooseMusic() {
        JFileChooser chooser = new JFileChooser();
        chooser.setDialogTitle("Choose Music (WAV, AIFF, or AU)");
        if (chooser.showOpenDialog(this) != JFileChooser.APPROVE_OPTION) return;
        try {
            music.load(chooser.getSelectedFile());
            musicStatus.setText(chooser.getSelectedFile().getName());
            music.setVolume(volumeSlider.getValue());
        } catch (Exception ex) {
            JOptionPane.showMessageDialog(this, "Could not load audio. Use WAV, AIFF, or AU.\n" + ex.getMessage(),
                    "Music Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    private void undo() {
        if (playing.get() || recordingActive.get()) return;
        movements.pop();
        rebuildLog();
    }

    private void clear() {
        if (recordingActive.get() || playing.get()) {
            JOptionPane.showMessageDialog(this, "Stop recording and playback first.");
            return;
        }
        movements.clear();
        rebuildLog();
        timerLabel.setText("00:00.000");
        lastAction.setText("Awaiting input");
    }

    private void rebuildLog() {
        logModel.clear();
        List<Movement> copy = movements.snapshot();
        for (Movement movement : copy) addLogLine(movement);
        count.setText(copy.size() + " events");
        if (copy.isEmpty()) {
            lastAction.setText("Awaiting input");
        } else {
            Movement last = copy.get(copy.size() - 1);
            lastAction.setText(last.key + (last.pressed ? " DOWN" : " UP"));
        }
        refreshButtons();
    }

    private void refreshButtons() {
        boolean busy = playing.get() || recordingActive.get();
        playButton.setEnabled(!playing.get() && !movements.isEmpty());
        stopButton.setEnabled(playing.get() || recordingActive.get());
        saveButton.setEnabled(!busy && !movements.isEmpty());
        loadButton.setEnabled(!busy);
        undoButton.setEnabled(!busy && !movements.isEmpty());
        clearButton.setEnabled(!busy);
        recordButton.setEnabled(!playing.get());
    }

    private void shutdown() {
        closing = true;
        stopPlayback.set(true);
        if (recordingActive.get()) stopRecording();
        music.close();
        if (hookRegistered) {
            try {
                GlobalScreen.removeNativeKeyListener(this);
                if (GlobalScreen.isNativeHookRegistered()) GlobalScreen.unregisterNativeHook();
            } catch (NativeHookException | RuntimeException ignored) {
                // The application is already closing; no further action is needed.
            }
        }
        dispose();
        System.exit(0);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            Main app = new Main();
            app.setVisible(true);
            Timer timer = new Timer(100, e -> app.updateTimer());
            timer.start();
        });
    }

    /* Built-in Java Sound player. Java Sound commonly supports WAV, AIFF and AU, not MP3 by default. */
    private static class MusicController {
        private Clip clip;
        private long pausedAt;
        private boolean paused;

        public void load(File file) throws Exception {
            close();
            try (AudioInputStream stream = AudioSystem.getAudioInputStream(file)) {
                clip = AudioSystem.getClip();
                clip.open(stream);
            }
            pausedAt = 0;
            paused = false;
        }

        public void playOrResume() {
            if (clip == null) return;
            if (paused) clip.setMicrosecondPosition(pausedAt);
            else if (!clip.isRunning() && clip.getMicrosecondPosition() >= clip.getMicrosecondLength()) {
                clip.setMicrosecondPosition(0);
            }
            paused = false;
            clip.start();
        }

        public void pause() {
            if (clip != null && clip.isRunning()) {
                pausedAt = clip.getMicrosecondPosition();
                clip.stop();
                paused = true;
            }
        }

        public void stop() {
            if (clip != null) {
                clip.stop();
                clip.setMicrosecondPosition(0);
                pausedAt = 0;
                paused = false;
            }
        }

        public void setVolume(int value) {
            if (clip == null || !clip.isControlSupported(FloatControl.Type.MASTER_GAIN)) return;
            FloatControl control = (FloatControl) clip.getControl(FloatControl.Type.MASTER_GAIN);
            float percent = Math.max(0, Math.min(100, value)) / 100.0f;
            float gain = percent == 0 ? control.getMinimum() : (float) (20.0 * Math.log10(percent));
            control.setValue(Math.max(control.getMinimum(), Math.min(control.getMaximum(), gain)));
        }

        public void close() {
            if (clip != null) {
                clip.stop();
                clip.close();
                clip = null;
            }
            pausedAt = 0;
            paused = false;
        }
    }
