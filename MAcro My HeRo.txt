    private void installKeyboardHook() {

        Logger.getLogger(
                GlobalScreen.class.getPackage().getName())
                .setLevel(Level.OFF);

        try {
            GlobalScreen.registerNativeHook();
            GlobalScreen.addNativeKeyListener(this);
        } catch (NativeHookException e) {
            status.setText("HOOK ERROR");

            JOptionPane.showMessageDialog(this,
                    "Keyboard capture could not start:\n"
                    + e.getMessage(),
                    "Keyboard Error",
                    JOptionPane.ERROR_MESSAGE);
        }
    }

    private String getKey(int code) {

        switch (code) {
            case NativeKeyEvent.VC_W: return "W";
            case NativeKeyEvent.VC_A: return "A";
            case NativeKeyEvent.VC_S: return "S";
            case NativeKeyEvent.VC_D: return "D";
            case NativeKeyEvent.VC_J: return "J";
            case NativeKeyEvent.VC_R: return "R";
            case NativeKeyEvent.VC_Q: return "Q";
            case NativeKeyEvent.VC_F: return "F";
            default: return null;
        }
    }

    private String getAction(String key) {

        switch (key) {
            case "W": return "MOVE UP";
            case "A": return "MOVE LEFT";
            case "S": return "MOVE DOWN";
            case "D": return "MOVE RIGHT";
            case "J": return "JUMP";
            case "R": return "RUN";
            case "Q": return "DODGE";
            case "F": return "BLOCK";
            default: return "UNKNOWN";
        }
    }

    @Override
    public void nativeKeyPressed(NativeKeyEvent event) {

        int code = event.getKeyCode();

        if (code == NativeKeyEvent.VC_F9) {
            SwingUtilities.invokeLater(this::toggleRecording);
            return;
        }

        if (code == NativeKeyEvent.VC_F10) {
            SwingUtilities.invokeLater(() -> playMacro(false));
            return;
        }

        String key = getKey(code);

        if (key == null || !recordingActive.get()) {
            return;
        }

        if (!heldKeys.add(code)) {
            return;
        }

        long now = System.nanoTime();
        long elapsed = now - recordingStart;

        LocalDateTime timestamp = LocalDateTime.now();

        synchronized (recording) {

            if (lastTimestamp != null
                    && !timestamp.isAfter(lastTimestamp)) {

                timestamp = lastTimestamp.plusNanos(1);
            }

            lastTimestamp = timestamp;

            Movement movement = new Movement(
                    code,
                    key,
                    getAction(key),
                    elapsed,
                    timestamp);

            recording.add(movement);

            SwingUtilities.invokeLater(
                    () -> addToLog(movement));
        }
    }

    @Override
    public void nativeKeyReleased(NativeKeyEvent event) {
        heldKeys.remove(event.getKeyCode());
    }

    @Override
    public void nativeKeyTyped(NativeKeyEvent event) {
    }

    private void addToLog(Movement movement) {

        String time = FORMAT.format(movement.timestamp);

        logModel.addElement(
                String.format("%-2s  %-10s  %s",
                        movement.key,
                        movement.action,
                        time));

        activityList.ensureIndexIsVisible(logModel.size() - 1);

        count.setText(String.valueOf(recording.size()));

        lastAction.setText(
                movement.key + " / " + movement.action);

        updateTimer();
        refreshButtons();
    }

    private void toggleRecording() {

        if (playing.get()) return;

        if (recordingActive.get()) {

            recordingActive.set(false);
            status.setText("STANDBY");
            recordButton.setText("● START RECORDING");
            recordButton.setBackground(RED);

        } else {

            recordingStart = System.nanoTime();
            recordingActive.set(true);

            status.setText("RECORDING");
            recordButton.setText("■ STOP RECORDING");
            recordButton.setBackground(new Color(130, 20, 30));
        }

        refreshButtons();
    }

    private void updateTimer() {

        if (!recordingActive.get()) return;

        long elapsed =
                (System.nanoTime() - recordingStart) / 1_000_000;

        long minutes = elapsed / 60000;
        long seconds = (elapsed / 1000) % 60;
        long millis = elapsed % 1000;

        timerLabel.setText(String.format(
                "%02d:%02d.%03d",
                minutes, seconds, millis));
    }

    private void playMacro(boolean countdown) {

        if (recordingActive.get()) {
            recordingActive.set(false);
            recordButton.setText("● START RECORDING");
            recordButton.setBackground(RED);
        }

        if (!playing.compareAndSet(false, true)) return;

        List<Movement> copy;

        synchronized (recording) {
            copy = new ArrayList<>(recording);
        }

        if (copy.isEmpty()) {
            playing.set(false);

            JOptionPane.showMessageDialog(this,
                    "Record some movements first.");

            refreshButtons();
            return;
        }

        status.setText("PLAYING");
        refreshButtons();

        Thread playback = new Thread(() -> {

            try {

                if (countdown) {
                    for (int i = 3; i > 0 && !closing; i--) {
                        int seconds = i;

                        SwingUtilities.invokeLater(
                                () -> status.setText(
                                        "SWITCH IN " + seconds));

                        Thread.sleep(1000);
                    }
                }

                Robot robot = new Robot();

                long previous = copy.get(0).elapsedNanos;

                for (int i = 0; i < copy.size() && !closing; i++) {

                    Movement movement = copy.get(i);

                    if (i > 0) {
                        long delay =
                                movement.elapsedNanos - previous;

                        if (delay > 0) {
                            long millis = delay / 1_000_000;
                            int nanos = (int) (delay % 1_000_000);

                            Thread.sleep(millis, nanos);
                        }
                    }

                    int keyCode = KeyEvent.getExtendedKeyCodeForChar(
                            movement.key.charAt(0));

                    robot.keyPress(keyCode);
                    robot.delay(45);
                    robot.keyRelease(keyCode);

                    previous = movement.elapsedNanos;
                }

            } catch (Exception e) {

                SwingUtilities.invokeLater(() ->
                        JOptionPane.showMessageDialog(this,
                                "Playback error: " + e.getMessage()));

            } finally {

                playing.set(false);

                SwingUtilities.invokeLater(() -> {
                    status.setText("STANDBY");
                    refreshButtons();
                });
            }

        }, "HeroMacroPlayback");

        playback.setDaemon(true);
        playback.start();
    }

    private void undo() {

        synchronized (recording) {

            if (recording.isEmpty() || playing.get()) return;

            recording.remove(recording.size() - 1);
            rebuildLog();
        }
    }

    private void clear() {

        if (recordingActive.get() || playing.get()) {
            JOptionPane.showMessageDialog(this,
                    "Stop recording and playback first.");
            return;
        }

        synchronized (recording) {
            recording.clear();
            lastTimestamp = null;
        }

        logModel.clear();
        count.setText("0");
        timerLabel.setText("00:00.000");
        lastAction.setText("Awaiting input");

        refreshButtons();
    }

    private void rebuildLog() {

        logModel.clear();

        for (Movement movement : recording) {

            logModel.addElement(
                    String.format("%-2s  %-10s  %s",
                            movement.key,
                            movement.action,
                            FORMAT.format(movement.timestamp)));
        }

        count.setText(String.valueOf(recording.size()));

        if (recording.isEmpty()) {
            lastAction.setText("Awaiting input");
        } else {
            Movement last = recording.get(recording.size() - 1);

            lastAction.setText(
                    last.key + " / " + last.action);
        }

        refreshButtons();
    }

    private void refreshButtons() {

        playButton.setEnabled(
                !playing.get() && !recording.isEmpty());

        undoButton.setEnabled(
                !playing.get() && !recording.isEmpty());

        clearButton.setEnabled(
                !playing.get() && !recordingActive.get());
    }

    private void shutdown() {

        closing = true;
        recordingActive.set(false);

        try {
            if (GlobalScreen.isNativeHookRegistered()) {
                GlobalScreen.unregisterNativeHook();
            }
        } catch (NativeHookException ignored) {
        }

        dispose();
        System.exit(0);
    }

        SwingUtilities.invokeLater(() -> {

            Main app = new Main();
            app.setVisible(true);

            Timer timer = new Timer(100,
                    e -> app.updateTimer());

            timer.start();
        });
    }
