import java.util.Stack;

public class DynamicStack {

    public static void main(String[] args) {

        Stack<String> recordedStack = new Stack<String>();
        Stack<String> playbackStack = new Stack<String>();

        // Record movements
        recordedStack.push("Run");
        recordedStack.push("Jump");
        recordedStack.push("Right");
        recordedStack.push("Attack");

        System.out.println("Recording movements:");
        System.out.println("Run");
        System.out.println("Jump");
        System.out.println("Right");
        System.out.println("Attack");

        // Transfer movements to playback stack
        while (!recordedStack.empty()) {
            playbackStack.push(recordedStack.pop());
        }

        // Play movements in original order
        System.out.println("\nPlaying back movements:");

        while (!playbackStack.empty()) {
            System.out.println(playbackStack.pop());
        }
    }
}
