Option Explicit

Dim stack() As String
Dim top As Integer

Sub StartStack()

    top = 0
    ReDim stack(1 To 1)

    MsgBox "Stack is ready!"

End Sub

Sub PushMovement()

    Dim movement As String

    movement = InputBox("Enter movement: Run, Jump, Left, Right, or Attack")

    If movement = "" Then Exit Sub

    top = top + 1

    ReDim Preserve stack(1 To top)

    stack(top) = movement

    MsgBox movement & " added to stack."

End Sub

Sub PopMovement()

    If top = 0 Then

        MsgBox "Stack is empty!"

    Else

        MsgBox "Removing: " & stack(top)

        top = top - 1

        If top > 0 Then
            ReDim Preserve stack(1 To top)
        Else
            Erase stack
        End If

    End If

End Sub

Sub ShowStack()

    Dim i As Integer
    Dim result As String

    If top = 0 Then

        MsgBox "Stack is empty!"

        Exit Sub

    End If

    result = "Current Stack:" & vbNewLine & vbNewLine

    For i = top To 1 Step -1

        result = result & stack(i) & vbNewLine

    Next i

    MsgBox result

End Sub
